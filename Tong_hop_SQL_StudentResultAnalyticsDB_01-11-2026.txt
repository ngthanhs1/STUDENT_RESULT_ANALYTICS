05. TẠO BẢNG DBO.KETQUA
============================================================
Chức năng: Lưu điểm từng sinh viên cho từng môn trong từng học kỳ.
Cột:
  MaKetQua       INT IDENTITY(1,1)   Khóa chính tự tăng
  MaSV           VARCHAR(20)         FK SinhVien
  MaMonHoc       VARCHAR(20)         FK MonHoc
  MaHocKy        VARCHAR(20)         FK HocKy
  DiemChuyenCan  DECIMAL(4,2)        Điểm 0..10 hoặc NULL
  DiemGiuaKy     DECIMAL(4,2)        Điểm 0..10 hoặc NULL
  DiemCuoiKy     DECIMAL(4,2)        Điểm 0..10 hoặc NULL
  DiemTongKet    DECIMAL(4,2)        Cột tự động tính, PERSISTED

Công thức: DiemTongKet = 0.1*DiemChuyenCan
                          + 0.3*DiemGiuaKy
                          + 0.6*DiemCuoiKy
Nếu một thành phần điểm là NULL thì điểm tổng kết cũng là NULL.
Khóa UNIQUE (MaSV, MaMonHoc, MaHocKy) ngăn nhập hai kết quả trùng bộ ba.

[BẮT ĐẦU SQL]
USE StudentResultAnalyticsDB;
GO

CREATE TABLE dbo.KetQua
(
    MaKetQua INT IDENTITY(1,1) PRIMARY KEY,

    MaSV VARCHAR(20) NOT NULL,
    MaMonHoc VARCHAR(20) NOT NULL,
    MaHocKy VARCHAR(20) NOT NULL,

    DiemChuyenCan DECIMAL(4,2) NULL,
    DiemGiuaKy DECIMAL(4,2) NULL,
    DiemCuoiKy DECIMAL(4,2) NULL,

    DiemTongKet AS
    CAST(ROUND(
        DiemChuyenCan * 0.1 +
        DiemGiuaKy * 0.3 +
        DiemCuoiKy * 0.6, 2
    ) AS DECIMAL(4,2)) PERSISTED,

    CONSTRAINT FK_KetQua_SinhVien
        FOREIGN KEY (MaSV)
        REFERENCES dbo.SinhVien(MaSV),

    CONSTRAINT FK_KetQua_MonHoc
        FOREIGN KEY (MaMonHoc)
        REFERENCES dbo.MonHoc(MaMonHoc),

    CONSTRAINT FK_KetQua_HocKy
        FOREIGN KEY (MaHocKy)
        REFERENCES dbo.HocKy(MaHocKy),

    CONSTRAINT UQ_KetQua
        UNIQUE (MaSV, MaMonHoc, MaHocKy),

    CONSTRAINT CK_KetQua_Diem
        CHECK (
            (DiemChuyenCan BETWEEN 0 AND 10
                OR DiemChuyenCan IS NULL)
            AND
            (DiemGiuaKy BETWEEN 0 AND 10
                OR DiemGiuaKy IS NULL)
            AND
            (DiemCuoiKy BETWEEN 0 AND 10
                OR DiemCuoiKy IS NULL)
        )
);
GO
[KẾT THÚC SQL]

06. TẠO BẢNG DBO.TAIKHOAN
============================================================
Chức năng: Chuẩn bị chức năng đăng nhập và phân quyền trong WPF.
Cột:
  MaTaiKhoan   INT IDENTITY(1,1)  Khóa chính

  MatKhauHash  NVARCHAR(255)      Dự kiến lưu chuỗi hash mật khẩu
  VaiTro       VARCHAR(20)        Admin / NhanVien
  TrangThai    BIT                1 hoạt động, 0 khóa
  NgayTao      DATETIME2(0)       UTC mặc định

[BẮT ĐẦU SQL]
USE StudentResultAnalyticsDB;
GO

CREATE TABLE dbo.TaiKhoan
(
    MaTaiKhoan INT IDENTITY(1,1) PRIMARY KEY,

    TenDangNhap VARCHAR(50) NOT NULL,
    MatKhauHash NVARCHAR(255) NOT NULL,

    VaiTro VARCHAR(20) NOT NULL
        DEFAULT 'NhanVien',

07. TẠO BẢNG DBO.LICHSUNHAPDULIEU
============================================================
Chức năng: Theo dõi việc nhập dữ liệu CSV/XLS/XLSX trong ứng dụng tương lai.
Cột:
  MaLichSu, TenFile, LoaiFile, ThoiGianNhap,
  TongSoDong, SoDongThanhCong, SoDongLoi,
  TrangThai, GhiChu, MaTaiKhoan (FK, cho phép NULL).

[BẮT ĐẦU SQL]
USE StudentResultAnalyticsDB;
GO

CREATE TABLE dbo.LichSuNhapDuLieu
(
    MaLichSu INT IDENTITY(1,1) PRIMARY KEY,

    TenFile NVARCHAR(255) NOT NULL,
    LoaiFile VARCHAR(10) NOT NULL,

    ThoiGianNhap DATETIME2(0) NOT NULL
        DEFAULT SYSDATETIME(),

    TongSoDong INT NOT NULL DEFAULT 0,
    SoDongThanhCong INT NOT NULL DEFAULT 0,
    SoDongLoi INT NOT NULL DEFAULT 0,

    TrangThai VARCHAR(20) NOT NULL
        DEFAULT 'ThanhCong',

    GhiChu NVARCHAR(500) NULL,
    MaTaiKhoan INT NULL,

    CONSTRAINT FK_LichSuNhap_TaiKhoan
        FOREIGN KEY (MaTaiKhoan)
        REFERENCES dbo.TaiKhoan(MaTaiKhoan),

    CONSTRAINT CK_LichSuNhap_LoaiFile
        CHECK (LoaiFile IN ('CSV', 'XLS', 'XLSX')),

    CONSTRAINT CK_LichSuNhap_TrangThai
        CHECK (TrangThai IN
            ('ThanhCong', 'MotPhan', 'Loi')),

    CONSTRAINT CK_LichSuNhap_SoDong
        CHECK (
            TongSoDong >= 0
            AND SoDongThanhCong >= 0
            AND SoDongLoi >= 0
            AND SoDongThanhCong + SoDongLoi <= TongSoDong
        )
);
GO
[KẾT THÚC SQL]

10. NHẬP 9 KẾT QUẢ HỌC TẬP MẪU
============================================================
Đã thực hiện 9 dòng cho 3 sinh viên x 3 môn trong HK1 2025-2026.
Riêng SV003 - MH003 thiếu DiemGiuaKy (NULL) để thử cảnh báo.

[BẮT ĐẦU SQL]
USE StudentResultAnalyticsDB;
GO

-- NHAP DU LIEU KET QUA HOC TAP
INSERT INTO dbo.KetQua
    (MaSV, MaMonHoc, MaHocKy,
     DiemChuyenCan, DiemGiuaKy, DiemCuoiKy)

SELECT
    d.MaSV, d.MaMonHoc, d.MaHocKy,
    d.DiemChuyenCan, d.DiemGiuaKy, d.DiemCuoiKy
FROM (VALUES
    ('SV001', 'MH001', 'HK1_2025_2026', 9.0, 8.0, 9.0),
    ('SV001', 'MH002', 'HK1_2025_2026', 8.0, 7.0, 8.0),
    ('SV001', 'MH003', 'HK1_2025_2026', 9.0, 9.0, 9.0),

    ('SV002', 'MH001', 'HK1_2025_2026', 7.0, 6.0, 7.0),
    ('SV002', 'MH002', 'HK1_2025_2026', 6.0, 5.0, 6.0),
    ('SV002', 'MH003', 'HK1_2025_2026', 8.0, 7.0, 8.0),

    ('SV003', 'MH001', 'HK1_2025_2026', 5.0, 4.0, 4.0),
    ('SV003', 'MH002', 'HK1_2025_2026', 6.0, 4.0, 5.0),
    ('SV003', 'MH003', 'HK1_2025_2026', 5.0, NULL, 6.0)
) AS d
    (MaSV, MaMonHoc, MaHocKy,
     DiemChuyenCan, DiemGiuaKy, DiemCuoiKy)

WHERE NOT EXISTS
(
    SELECT 1
    FROM dbo.KetQua AS k
    WHERE k.MaSV = d.MaSV
      AND k.MaMonHoc = d.MaMonHoc
      AND k.MaHocKy = d.MaHocKy
);
GO

-- KIEM TRA KET QUA
SELECT
    sv.MaSV,
    sv.HoTen,
    mh.TenMonHoc,
    mh.SoTinChi,
    k.DiemChuyenCan,
    k.DiemGiuaKy,
    k.DiemCuoiKy,

11. TRUY VẤN TÍNH ĐIỂM TRUNG BÌNH THEO SỐ TÍN CHỈ
============================================================
Công thức hệ 10:
  Điểm TB = SUM(DiemTongKet * SoTinChi) / SUM(SoTinChi có điểm)
Bỏ môn còn NULL khi tính điểm TẠM TÍNH; đếm riêng môn thiếu điểm.
Đây KHÔNG phải GPA hệ 4.

[BẮT ĐẦU SQL]
USE StudentResultAnalyticsDB;
GO

-- TINH DIEM TRUNG BINH THEO TIN CHI
SELECT
    sv.MaSV,
    sv.HoTen,
    k.MaHocKy,

    COUNT(*) AS TongSoMon,

    SUM(CASE
        WHEN k.DiemTongKet IS NULL THEN 1
        ELSE 0
    END) AS SoMonThieuDiem,

    SUM(CASE
        WHEN k.DiemTongKet IS NOT NULL
        THEN mh.SoTinChi
        ELSE 0
    END) AS SoTinChiDaCoDiem,

    CAST(
        SUM(CASE
            WHEN k.DiemTongKet IS NOT NULL
            THEN k.DiemTongKet * mh.SoTinChi
            ELSE 0
        END) * 1.0
        /
        NULLIF(SUM(CASE
            WHEN k.DiemTongKet IS NOT NULL
            THEN mh.SoTinChi
            ELSE 0
        END), 0)
        AS DECIMAL(5,2)
    ) AS DiemTrungBinhTamTinh

FROM dbo.KetQua AS k
JOIN dbo.SinhVien AS sv
    ON k.MaSV = sv.MaSV
JOIN dbo.MonHoc AS mh
    ON k.MaMonHoc = mh.MaMonHoc

WHERE k.MaHocKy = 'HK1_2025_2026'

GROUP BY sv.MaSV, sv.HoTen, k.MaHocKy
ORDER BY sv.MaSV;

  SV002 - Trần Thị Bình: 6.58 | 8 tín chỉ có điểm | 0 môn thiếu

                ELSE 0

            /
            NULLIF(SUM(CASE

                ELSE 0

HẠN CHẾ CẦN BIẾT KHI PHÁT TRIỂN TIẾP
- Các VIEW hiện thống kê bản ghi đã có trong KetQua; không tự xác định
  một sinh viên chưa có bản ghi kết quả cho môn chưa đăng ký.
- Điểm 4.45 của SV003 chỉ là tạm tính vì đang thiếu điểm giữa kỳ MH003.
- Bảng TaiKhoan đã có 2 tài khoản mẫu (admin, nhanvien) sẵn sàng cho màn hình Login trong WPF.
- Bảng LichSuNhapDuLieu đã có cấu trúc nhưng chưa có lịch sử nhập thật.
- Quy tắc phân loại điểm và ngưỡng đậu/trượt cần xác nhận theo quy chế
  thực tế nếu sử dụng cho dữ liệu của một trường học cụ thể.
